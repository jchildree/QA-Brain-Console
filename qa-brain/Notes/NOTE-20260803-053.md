# SOP Vertical Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first vertical slice of the new QA-Brain console (per ADR-002, which supersedes ADR-001's "dashboard app" framing) for the SOP record type only: a persistent-nav console shell, a cross-type overview landing screen, a status dashboard replacing `qa-brain/_dashboards/SOP-Control.md`'s Dataview queries, a schema-driven form for editing SOP frontmatter, a Claude-assisted intake flow that extracts frontmatter fields from pasted free text, an extract-and-attach evidence file upload, and stub screens for the 5 record types plus 3 utility screens not yet built.

**Architecture:** Two local Node processes: an Express backend (`app/backend`) that is the only thing that touches the filesystem and the Anthropic API (holds `ANTHROPIC_API_KEY` server-side, reads/writes `qa-brain/SOP/*.md` frontmatter via `gray-matter`), and a Vite + React + TypeScript frontend (`app/frontend`) that talks to the backend over `fetch`. No database --  the `.md` files in `qa-brain/SOP/` remain the single source of truth, read fresh on every request and written only when the user saves in the app (no file watcher, no live sync). The SOP body (sections 1-9 of the template) is edited as a single Markdown textarea in this slice; only frontmatter fields are schema-driven --  exploding the body into per-section structured fields is out of scope here. The frontend is a console shell (`AppShell`) with a persistent nav across 10 sections: Overview (cross-type landing screen, real SOP counts + stub rows for the other 5 types), SOP (dashboard/editor/intake, fully functional), 5 stub record-type screens (CAP, Incident, Change, Audit, Training), and 3 stub utility screens (Settings, Evidence Library, Search) -- all 8 stubs render placeholder-only content this pass. `qa-brain/_dashboards/Home.md` gets a one-line link to the new console.

**Tech Stack:** Node 24 / Express 4 / `gray-matter` / `@anthropic-ai/sdk` / `multer` --  backend. Vite / React 18 / TypeScript / `react-router-dom` --  frontend. `vitest` + `supertest` for backend tests, `vitest` + `@testing-library/react` for frontend tests.

const fs = require('fs');
const path = require('path');
const matter = require('gray-matter');

function listSops(sopDir) {
  return fs
    .readdirSync(sopDir)
    .filter((f) => f.endsWith('.md'))
    .map((f) => {
      const raw = fs.readFileSync(path.join(sopDir, f), 'utf8');
      const { data } = matter(raw);
      return { ...data, filename: f };
    });
}

function readSop(sopDir, filename) {
  const filePath = path.join(sopDir, filename);
  const raw = fs.readFileSync(filePath, 'utf8');
  const { data, content } = matter(raw);
  return { frontmatter: data, body: content, filename };
}

function writeSop(sopDir, filename, frontmatter, body) {
  const filePath = path.join(sopDir, filename);
  const output = matter.stringify(body, frontmatter);
  fs.writeFileSync(filePath, output, 'utf8');
}

module.exports = { listSops, readSop, writeSop };
```

const { listSops, readSop, writeSop } = require('../vault');

function createSopsRouter(sopDir) {
  const router = express.Router();

  router.get('/sops', (req, res) => {
    res.json(listSops(sopDir));
  });

  router.get('/sops/:filename', (req, res) => {
    try {
      res.json(readSop(sopDir, req.params.filename));
    } catch (err) {
      res.status(404).json({ error: 'SOP not found' });
    }
  });

  router.put('/sops/:filename', (req, res) => {
    const { frontmatter, body } = req.body;
    writeSop(sopDir, req.params.filename, frontmatter, body);
    res.json({ ok: true });
  });

  return router;
}

module.exports = { createSopsRouter };

async function extractSopFields(client, rawText) {
  const message = await client.messages.create({
    model: 'claude-sonnet-5',
    max_tokens: 1024,
    messages: [
      {
        role: 'user',
        content: `Extract SOP frontmatter fields from this raw text. Return ONLY valid JSON with keys: department (one of QC, QA, IT, Operations), tags (array of strings), purpose (short string), scope (short string). Raw text:\n\n${rawText}`,
      },
    ],
  });
  const textBlock = message.content.find((block) => block.type === 'text');
  return JSON.parse(textBlock.text);
}

module.exports = { extractSopFields };

const express = require('express');
const { extractSopFields } = require('../claude');

function createIntakeRouter(anthropicClient) {
  const router = express.Router();

  router.post('/intake/extract', async (req, res) => {
    const { rawText } = req.body;
    if (!rawText || !rawText.trim()) {
      return res.status(400).json({ error: 'rawText is required' });
    }
    try {
      const extracted = await extractSopFields(anthropicClient, rawText);
      res.json(extracted);
    } catch (err) {
      res.status(502).json({ error: 'Claude extraction failed', detail: err.message });
    }
  });

  return router;
}

module.exports = { createIntakeRouter };
```

```js
const express = require('express');
const fs = require('fs');
const path = require('path');
const multer = require('multer');
const { extractSopFields } = require('../claude');

function createAttachmentsRouter(sopDir, anthropicClient) {
  const router = express.Router();
  const upload = multer({ storage: multer.memoryStorage() });

  router.post('/sops/:filename/attachments', upload.single('evidence'), async (req, res) => {
    if (!req.file) {
      return res.status(400).json({ error: 'evidence file is required' });
    }

    const sopBase = path.basename(req.params.filename, '.md');
    const attachDir = path.join(sopDir, 'attachments', sopBase);
    fs.mkdirSync(attachDir, { recursive: true });
    const destPath = path.join(attachDir, req.file.originalname);
    fs.writeFileSync(destPath, req.file.buffer);

    let extracted = null;
    let note = null;

    if (req.file.mimetype.startsWith('text/')) {
      try {
        extracted = await extractSopFields(anthropicClient, req.file.buffer.toString('utf8'));
      } catch (err) {
        note = `Extraction failed: ${err.message}`;
      }
    } else {
      note = 'Binary evidence attached; no automatic extraction for this file type';
    }

    res.json({ attachmentPath: destPath, extracted, note });
  });

  return router;
}

module.exports = { createAttachmentsRouter };

const BASE_URL = 'http://localhost:4000/api';

export interface SopSummary {
  id: string;
  status: string;
  version: string;
  department: string;
  review_date: string;
  owner: string;
  filename: string;
}

export interface SopDetail {
  frontmatter: Record<string, unknown>;
  body: string;
  filename: string;
}

export async function listSops(): Promise<SopSummary[]> {
  const res = await fetch(`${BASE_URL}/sops`);
  if (!res.ok) throw new Error('Failed to load SOPs');
  return res.json();
}

export async function getSop(filename: string): Promise<SopDetail> {
  const res = await fetch(`${BASE_URL}/sops/${filename}`);
  if (!res.ok) throw new Error('Failed to load SOP');
  return res.json();
}

export async function saveSop(filename: string, frontmatter: Record<string, unknown>, body: string): Promise<void> {
  const res = await fetch(`${BASE_URL}/sops/${filename}`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ frontmatter, body }),
  });
  if (!res.ok) throw new Error('Failed to save SOP');
}

export async function extractFromText(rawText: string): Promise<Record<string, unknown>> {
  const res = await fetch(`${BASE_URL}/intake/extract`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ rawText }),
  });
  if (!res.ok) throw new Error('Extraction failed');
  return res.json();
}

export async function uploadAttachment(filename: string, file: File): Promise<{ attachmentPath: string; extracted: Record<string, unknown> | null; note: string | null }> {
  const formData = new FormData();
  formData.append('evidence', file);
  const res = await fetch(`${BASE_URL}/sops/${filename}/attachments`, {
    method: 'POST',
    body: formData,
  });
  if (!res.ok) throw new Error('Attachment upload failed');
  return res.json();
}

interface Props {
  blueprint: BlueprintField[];
  values: Record<string, unknown>;
  onChange: (key: string, value: unknown) => void;
}

export function DynamicForm({ blueprint, values, onChange }: Props) {
  return (
    <div className="dynamic-form">
      {blueprint.map((field) => {
        const labelText = `${field.label}${field.required ? ' *' : ''}`;
        return (
          <label key={field.key} className="dynamic-form-field">
            <span>{labelText}</span>
            {field.type === 'select' ? (
              <select aria-label={labelText} value={(values[field.key] as string) ?? ''} onChange={(e) => onChange(field.key, e.target.value)}>
                <option value="">-- select --</option>
                {field.options?.map((opt) => (
                  <option key={opt} value={opt}>
                    {opt}
                  </option>
                ))}
              </select>
            ) : field.type === 'tags' ? (
              <input
                aria-label={labelText}
                type="text"
                value={((values[field.key] as string[]) ?? []).join(', ')}
                onChange={(e) =>
                  onChange(
                    field.key,
                    e.target.value.split(',').map((s) => s.trim()).filter(Boolean)
                  )
                }
              />
            ) : (
              <input
                aria-label={labelText}
                type={field.type === 'date' ? 'date' : 'text'}
                value={(values[field.key] as string) ?? ''}
                onChange={(e) => onChange(field.key, e.target.value)}
              />
            )}
          </label>
        );
      })}

import { listSops, SopSummary } from '../api/client';
import { StatusPill } from '../components/StatusPill';

function isOverdue(sop: SopSummary): boolean {
  return sop.status === 'Approved' && new Date(sop.review_date) < new Date();
}

function SopTable({ sops }: { sops: SopSummary[] }) {
  return (
    <table>
      <thead>
        <tr>
          <th>ID</th>
          <th>Version</th>
          <th>Status</th>
          <th>Department</th>
          <th>Review Date</th>

            <td>{sop.owner}</td>

export function Dashboard() {

  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    listSops()

      .catch((err: Error) => setError(err.message));

  const overdue = sops.filter(isOverdue);
  const inDraftOrReview = sops.filter((s) => s.status === 'Draft' || s.status === 'In Review');

      </section>
      <section>

      </section>
      <section>
        <h2>In Draft / Review ({inDraftOrReview.length})</h2>
        <SopTable sops={inDraftOrReview} />
      </section>
      <section>
        <h2>Obsolete SOPs ({obsolete.length})</h2>
        <SopTable sops={obsolete} />
      </section>

import { useEffect, useState } from 'react';
import { getSop, saveSop } from '../api/client';
import { DynamicForm } from '../components/DynamicForm';
import { sopBlueprint } from '../blueprint/sopBlueprint';

export function SopEditor({ filename }: { filename: string }) {
  const [frontmatter, setFrontmatter] = useState<Record<string, unknown>>({});
  const [body, setBody] = useState('');
  const [loaded, setLoaded] = useState(false);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    getSop(filename).then((sop) => {
      setFrontmatter(sop.frontmatter);
      setBody(sop.body);
      setLoaded(true);
    });
  }, [filename]);

  if (!loaded) return <p>Loading…</p>;

  const handleChange = (key: string, value: unknown) => {
    setFrontmatter((prev) => ({ ...prev, [key]: value }));
  };

  const handleSave = async () => {
    setSaving(true);
    await saveSop(filename, frontmatter, body);
    setSaving(false);
  };

  return (
    <div className="sop-editor">
      <DynamicForm blueprint={sopBlueprint} values={frontmatter} onChange={handleChange} />
      <label className="sop-editor-body">
        <span>Body</span>
        <textarea aria-label="Body" value={body} onChange={(e) => setBody(e.target.value)} rows={20} />
      </label>
      <button type="button" onClick={handleSave} disabled={saving}>
        Save
      </button>
    </div>
  );
}
```

import { extractFromText } from '../api/client';

export function Intake() {

      <button type="button" onClick={handleExtract} disabled={!rawText.trim() || extracting}>
        Extract
      </button>
      <DynamicForm blueprint={sopBlueprint} values={fields} onChange={handleChange} />
    </div>
  );
}

import { useEffect, useState } from 'react';
import { listSops, SopSummary } from '../api/client';
import { isOverdue } from './Dashboard';

const STUB_TYPES = ['CAP', 'Incident', 'Change', 'Audit', 'Training'];

export function Overview() {
  const [sops, setSops] = useState<SopSummary[]>([]);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    listSops()
      .then(setSops)
      .catch((err: Error) => setError(err.message));
  }, []);

  if (error) return <p role="alert">{error}</p>;

  const overdue = sops.filter(isOverdue).length;

  return (
    <div className="overview">
      <h1>Console Overview</h1>
      <table>
        <thead>
          <tr>
            <th>Record Type</th>
            <th>Status</th>
          </tr>
        </thead>
        <tbody>
          <tr>
            <td>SOP</td>
            <td>
              {sops.length} total, {overdue} overdue for review
            </td>
          </tr>
          {STUB_TYPES.map((type) => (
            <tr key={type}>
              <td>{type}</td>
              <td>-- not yet available</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
